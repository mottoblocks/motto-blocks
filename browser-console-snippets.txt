// Subscribe to selection changes (unsubscribe with __unsubSel())
window.__unsubSel?.();
window.__unsubSel = wp.data.subscribe(() => {
  const b = wp.data.select('core/block-editor').getSelectedBlock();
  if (!b) return;
  const next = `${b.clientId}:${b.name}`;
  if (window.__lastSel === next) return;
  window.__lastSel = next;
  console.log('[selected]', { name: b.name, clientId: b.clientId, mottoStyler: b.attributes?.mottoStyler || b.attributes?.mottoProps });
});

// Subscribe to MottoStyler changes on selected block (unsubscribe with __unsubMotto())
window.__unsubMotto?.();
window.__lastMottoJson = '';
window.__unsubMotto = wp.data.subscribe(() => {
  const b = wp.data.select('core/block-editor').getSelectedBlock();
  const json = JSON.stringify(b?.attributes?.mottoStyler || b?.attributes?.mottoProps || {});
  if (json === window.__lastMottoJson) return;
  window.__lastMottoJson = json;
  console.log('[mottoStyler changed]', b?.name, b?.clientId, JSON.parse(json));
});
